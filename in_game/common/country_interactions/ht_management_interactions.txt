# ── Two-target: recipient = annexer, target = annexed ────────────────────────────

ht_annex = {
	type = diplomacy
	category = CATEGORY_HT_ACTIONS
	ai_tick = never
	use_enroute = no
	block_when_at_war = no
	show_message = no
	show_message_to_target = no

	potential = {
		scope:actor = { cmf_can_use_unrestricted_tools = yes }
	}

	allow = { }

	select_trigger = {
		allow_self = yes
		looking_for_a = country
		target_flag = recipient
		name = ht_select_first_country


		show_if = {
			NOT = { has_global_variable = ht_target_self }
		}

		column = { data = name }

		visible = {
			NOR = {
				country_type = building
				country_type = pop
			}
		}
	}

	select_trigger = {
		allow_self = yes
		looking_for_a = country
		target_flag = recipient
		name = ht_select_self_to_continue


		show_if = {
			has_global_variable = ht_target_self
			NOT = { country_exists = scope:recipient }
		}

		column = { data = name }

		visible = {
			this = scope:actor
		}
	}

	select_trigger = {
		allow_self = no
		looking_for_a = country
		target_flag = target
		name = ht_select_second_country


		show_if = {
			NOT = { has_global_variable = ht_target_self }
			country_exists = scope:recipient
		}

		column = { data = name }

		visible = {
			NOR = {
				country_type = building
				country_type = pop
			}
			NOT = { this = scope:recipient }
		}
	}

	effect = {
		hidden_effect = {
			if = {
				limit = { has_global_variable = ht_target_self }
				if = {
					limit = { country_exists = scope:recipient }
					scope:recipient = { save_scope_as = target }
				}
				scope:actor = { save_scope_as = recipient }
			}
			if = {
				limit = {
					exists = scope:recipient
					exists = scope:target
				}
				ht_annex_effect = yes
			}
		}
	}
}

# -- Single-target: recipient = country the land is given to --------------------

ht_give_land = {
	type = diplomacy
	category = CATEGORY_HT_ACTIONS
	ai_tick = never
	use_enroute = no
	block_when_at_war = no
	show_message = no
	show_message_to_target = no

	potential = {
		scope:actor = { cmf_can_use_unrestricted_tools = yes }
	}

	allow = { }

	select_trigger = {
		allow_self = yes
		looking_for_a = country
		target_flag = recipient
		name = ht_select_land_receiver


		show_if = {
			NOT = { has_global_variable = ht_target_self }
		}

		column = { data = name }

		visible = {
			NOR = {
				country_type = building
				country_type = pop
			}
		}

		enabled = {
			num_locations > 0
		}
	}

	select_trigger = {
		allow_self = yes
		looking_for_a = country
		target_flag = recipient
		name = ht_select_self_to_continue


		show_if = {
			has_global_variable = ht_target_self
			NOT = { country_exists = scope:recipient }
		}

		column = { data = name }

		visible = {
			this = scope:actor
		}

		enabled = {
			num_locations > 0
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = ht_size_choice
		name = ht_select_land_size

		interaction_source_list = {
			situation:ht_size_location = { add_to_list = source }
			situation:ht_size_province = { add_to_list = source }
			situation:ht_size_area = { add_to_list = source }
		}

		column = { data = name }
	}

	select_trigger = {
		looking_for_a = location
		target_flag = ht_target_location
		source_flags = only_actual_locations
		name = ht_select_location_to_give
		none_available_msg_key = ht_no_valid_locations

		interaction_source_list = {
			if = {
				limit = { has_global_variable = ht_target_self }
				ht_give_land_location_sources = { receiver = scope:actor }
			}
			else = {
				ht_give_land_location_sources = { receiver = scope:recipient }
			}
		}

		show_if = {
			exists = scope:ht_size_choice
			scope:ht_size_choice = situation:ht_size_location
		}

		column = { data = name }
		column = { data = population }

		visible = {
			OR = {
				AND = {
					has_global_variable = ht_target_self
					ht_can_give_location = { receiver = scope:actor }
				}
				AND = {
					NOT = { has_global_variable = ht_target_self }
					ht_can_give_location = { receiver = scope:recipient }
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		target_flag = ht_target_province
		name = ht_select_province_to_give
		none_available_msg_key = ht_no_valid_provinces

		interaction_source_list = {
			if = {
				limit = { has_global_variable = ht_target_self }
				ht_give_land_province_sources = { receiver = scope:actor }
			}
			else = {
				ht_give_land_province_sources = { receiver = scope:recipient }
			}
		}

		show_if = {
			exists = scope:ht_size_choice
			scope:ht_size_choice = situation:ht_size_province
		}

		column = { data = name }
		column = { data = population }

		visible = {
			OR = {
				AND = {
					has_global_variable = ht_target_self
					ht_can_give_province = { receiver = scope:actor }
				}
				AND = {
					NOT = { has_global_variable = ht_target_self }
					ht_can_give_province = { receiver = scope:recipient }
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = area
		target_flag = ht_target_area
		name = ht_select_area_to_give
		none_available_msg_key = ht_no_valid_areas

		interaction_source_list = {
			if = {
				limit = { has_global_variable = ht_target_self }
				ht_give_land_area_sources = { receiver = scope:actor }
			}
			else = {
				ht_give_land_area_sources = { receiver = scope:recipient }
			}
		}

		show_if = {
			exists = scope:ht_size_choice
			scope:ht_size_choice = situation:ht_size_area
		}

		column = { data = name }
		column = { data = population }

		visible = {
			OR = {
				AND = {
					has_global_variable = ht_target_self
					ht_can_give_area = { receiver = scope:actor }
				}
				AND = {
					NOT = { has_global_variable = ht_target_self }
					ht_can_give_area = { receiver = scope:recipient }
				}
			}
		}
	}

	effect = {
		hidden_effect = {
			if = {
				limit = { has_global_variable = ht_target_self }
				scope:actor = { save_scope_as = recipient }
			}
			if = {
				limit = {
					exists = scope:recipient
				}
				scope:actor = {
					set_variable = {
						name = ht_land_receiver
						value = scope:recipient
					}
				}
				ht_run_give_land = yes
			}
		}
	}
